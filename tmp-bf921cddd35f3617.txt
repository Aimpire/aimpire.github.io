var HUB = 'https://aimpire.github.io/';
var ARK = 'Skjemasvar 1';          // fanen skjemasvarene havner i
var KOL_GODKJENT = 'Approved';     // haken du setter
var KOL_SENDT = 'Mail sent';       // settes av skriptet, så ingen får to e-poster

function paaEndring(e) {
  if (!e || !e.range) return;
  var ark = e.range.getSheet();
  if (!erSoeknadsfane_(ark)) return;
  var rad = e.range.getRow();
  if (rad < 2) return;
  /* To haker satt rett etter hverandre = to kjoeringer samtidig. Laasen gjoer at den
     siste ikke overskriver «Publisert» med et gammelt bilde (gjennomgang 16.10, funn 5). */
  var laas = LockService.getScriptLock();
  laas.waitLock(30000);
  try { paaEndring_(e, ark, rad); } finally { laas.releaseLock(); }
}

function kolonne_(tittel, navn) {
  var c = tittel.indexOf(navn) + 1;
  if (!c) throw new Error('Mangler kolonnen «' + navn + '» i fanen «' + ARK + '» - overskriften maa staa akkurat slik.');
  return c;
}

/** Navn maa ha minst én bokstav (latinsk, arabisk/urdu eller devanagari) - ellers kaster siden raden. */
function navnOk_(navn) {
  /* Samme regel som siden: minst én bokstav i et hvilket som helst skriftsystem
     (ogsaa gurmukhi, bengali, kyrillisk) - gjennomgang 17.10, funn 4. */
  return /\p{L}/u.test(String(navn || ''));
}

/** Soeknadsfanen kjennes paa overskriftene, ikke bare paa navnet - et omdoept ark skal ikke tie (funn 7). */
function erSoeknadsfane_(ark) {
  if (ark.getName() === ARK) return true;
  var n = ark.getLastColumn(); if (n < 1) return false;
  var t = ark.getRange(1, 1, 1, n).getValues()[0];
  return t.indexOf(KOL_GODKJENT) >= 0 && t.indexOf('Partner link') >= 0;
}

function stempel_() {
  return Utilities.formatDate(new Date(), SpreadsheetApp.getActive().getSpreadsheetTimeZone(), 'dd.MM HH:mm');
}

function paaEndring_(e, ark, radStart) {
  var siste = ark.getLastColumn();
  var tittel = ark.getRange(1, 1, 1, siste).getValues()[0];
  var cGodkjent = kolonne_(tittel, KOL_GODKJENT);
  var cNavn = kolonne_(tittel, 'First name');
  var cLenke = kolonne_(tittel, 'Partner link');
  kolonne_(tittel, 'Email');
  var kol = e.range.getColumn(), antKol = e.range.getNumColumns();
  var treffer = function (c) { return c >= kol && c < kol + antKol; };

  /* «Publisert» skrives om som VERDIER ved hver endring i Approved, First name eller
     Partner link - ogsaa en retting av navnet paa en allerede godkjent rad (funn 1).
     Google publiserer fanen paa nytt bare naar den selv endres - en formel som leser
     denne fanen teller ikke (maalt 16.10: haken satt 17:11, CSV-en fortsatt tom 17:25). */
  if (!treffer(cGodkjent) && !treffer(cNavn) && !treffer(cLenke)) return;
  oppdaterPublisert_(ark, tittel, cGodkjent);

  var cSendt = tittel.indexOf(KOL_SENDT) + 1;
  if (!cSendt) {
    cSendt = siste + 1;
    ark.getRange(1, cSendt).setValue(KOL_SENDT);
  }
  /* Ny partnerlenke paa en godkjent rad = ny kode = ny Aimpire-lenke. Den gamle
     lenken lederen har delt, slutter aa virke. Merk raden, saa founder ser det og
     en ny hake sender ny e-post (gjennomgang 17.10, funn 2). */
  if (treffer(cLenke) && !treffer(cGodkjent)) {
    var antRadL = e.range.getNumRows();
    for (var rl = radStart; rl < radStart + antRadL; rl++) {
      var sendtL = ark.getRange(rl, cSendt).getValue();
      if (ark.getRange(rl, cGodkjent).getValue() === true && sendtL && !/^(FEIL|KVOTE|NY)/.test(String(sendtL))) {
        ark.getRange(rl, cSendt).setValue('NY LENKE ' + stempel_());
      }
    }
    return;
  }
  if (!treffer(cGodkjent)) return;
  /* Flere rader limt inn paa én gang: hver rad for seg (funn 8). */
  var antRad = e.range.getNumRows();
  for (var r = radStart; r < radStart + antRad; r++) {
    if (ark.getRange(r, cGodkjent).getValue() === true) sendTil_(ark, tittel, r, cSendt, siste);
  }
}

function sendTil_(ark, tittel, rad, cSendt, siste) {
  var sendt = ark.getRange(rad, cSendt).getValue();
  /* En dato = sendt. FEIL/KVOTE/NY LENKE = send (paa nytt) ved neste hake (funn 2). */
  if (sendt && !/^(FEIL|KVOTE|NY)/.test(String(sendt))) return;
  var merk = function (tekst) { ark.getRange(rad, cSendt).setValue(tekst + ' ' + stempel_()); };

  var verdier = ark.getRange(rad, 1, 1, Math.max(siste, cSendt)).getValues()[0];
  var navn = String(verdier[tittel.indexOf('First name')] || '').trim();
  var lenke = String(verdier[tittel.indexOf('Partner link')] || '').trim();
  var epost = String(verdier[tittel.indexOf('Email')] || '').trim();
  /* #49: bare partnerlenken slik IB Portal gir den. Feil format = ingen e-post,
     og «FEIL LENKE» i kolonnen saa founder ser det med en gang. */
  if (!LENKE_OK.test(lenke)) { merk('FEIL LENKE'); return; }
  if (!navnOk_(navn)) { merk('FEIL NAVN'); return; }
  if (!epost || epost.indexOf('@') < 1) { merk('FEIL E-POST'); return; }

  var min = HUB + '?l=' + kodeAv_(lenke);
  var kode = String(verdier[tittel.indexOf('Language')] || 'en').trim().toLowerCase();
  if (!TEKST[kode]) kode = 'en';
  var t = TEKST[kode];
  var emne = t.subject;
  /* Funksjon som erstatning, saa «$» i et navn ikke tolkes som moenster (funn 13). */
  var tekst = t.body.replace('{NAME}', function () { return navn; }).replace('{LINK}', function () { return min; });
  /* Vanlig Gmail har 100 e-poster/dag fra Apps Script. Er kvoten brukt opp,
     skrives KVOTE i kolonnen i stedet for en dato, saa founder ser det - lederen
     har uansett lenken fra verktoeyet, saa ingenting stopper. */
  if (MailApp.getRemainingDailyQuota() < 1) { merk('KVOTE'); return; }
  /* Avsendernavn «Aimpire» (ikke en naken gmail-adresse) - mindre spam-aktig,
     og lederen kjenner igjen hvem det er fra. */
  try {
    MailApp.sendEmail(epost, emne, tekst, { name: 'Aimpire' });
  } catch (feil) {
    merk('FEIL SENDING ' + String(feil && feil.message || feil).slice(0, 60));
    return;
  }
  ark.getRange(rad, cSendt).setValue(new Date());
}

/**
 * «Publisert»-fanen = de godkjente radene som verdier: fornavn | partnerlenke.
 * Overskriver hele fanen hver gang, saa en fjernet hake ogsaa fjerner lederen.
 * Siden leser den publiserte CSV-en av denne fanen.
 */
var ARK_PUBLISERT = 'Publisert';
/** Partnerlenken slik IB Portal -> Profile gir den. Samme regel som i skjemaet og paa siden. */
var LENKE_OK = /^https:\/\/(www\.)?copyxapp\.com\/auth\/register\?e=[A-Za-z0-9_%.+\/=-]{20,}&a=1$/;
function oppdaterPublisert_(ark, tittel, cGodkjent) {
  var pub = SpreadsheetApp.getActive().getSheetByName(ARK_PUBLISERT);
  if (!pub) throw new Error('Mangler fanen «' + ARK_PUBLISERT + '» - siden faar ingen ledere uten den.');
  var cNavn = kolonne_(tittel, 'First name'), cLenke = kolonne_(tittel, 'Partner link');
  var sisteRad = ark.getLastRow();
  var rader = [];
  if (sisteRad >= 2) {
    var alle = ark.getRange(2, 1, sisteRad - 1, ark.getLastColumn()).getValues();
    for (var i = 0; i < alle.length; i++) {
      var r = alle[i];
      var navn = String(r[cNavn - 1] || '').trim();
      if (r[cGodkjent - 1] === true && navnOk_(navn) && LENKE_OK.test(String(r[cLenke - 1] || '').trim())) {
        rader.push([navn, String(r[cLenke - 1]).trim()]);
      }
    }
  }
  pub.clearContents();
  if (rader.length) pub.getRange(1, 1, rader.length, 2).setValues(rader);
  SpreadsheetApp.flush();
}

/** Samme regnestykke som siden: FNV-1a over «vert|e-parameter», base36, 6 tegn. */
function kodeAv_(raa) {
  var d = delUrl_(raa);
  if (!d) return '';
  var e = parameter_(d.search, 'e');
  var s = d.hostname + '|' + (e || (d.pathname + d.search));
  var h = 0x811c9dc5;
  for (var i = 0; i < s.length; i++) {
    h ^= s.charCodeAt(i);
    h = Math.imul(h, 0x01000193) >>> 0;
  }
  return ('00000' + h.toString(36)).slice(-6);
}

function delUrl_(raa) {
  var m = String(raa).trim().match(/^https:\/\/([^\/?#]+)([^?#]*)(\?[^#]*)?/);
  if (!m) return null;
  return { hostname: m[1].toLowerCase(), pathname: m[2] || '/', search: m[3] || '' };
}

function parameter_(search, navn) {
  var biter = String(search).replace(/^\?/, '').split('&');
  for (var i = 0; i < biter.length; i++) {
    if (!biter[i]) continue;
    var j = biter[i].indexOf('=');
    var k = j < 0 ? biter[i] : biter[i].slice(0, j);
    if (dekod_(k) === navn) {
      return j < 0 ? '' : dekod_(biter[i].slice(j + 1));
    }
  }
  return null;
}

/** Som siden: «+» = mellomrom; en ugyldig %-sekvens gir raaverdien i stedet for feil (funn 6). */
function dekod_(t) {
  var u = String(t).replace(/\+/g, ' ');
  try { return decodeURIComponent(u); } catch (feil) { return u; }
}

/**
 * Lenken til en leder, regnet ut av partnerlenka. Brukes som vanlig formel i
 * arket, saa founder har alle lenkene paa ett sted:
 *     =ARRAYFORMULA(HVIS(C2:C="";"";AIMPIRELENKE(C2:C)))
 * @param {string|Array} lenke  partnerlenke, eller en kolonne av dem
 * @return {string|Array} den ferdige Aimpire-lenka
 * @customfunction
 */
function AIMPIRELENKE(lenke) {
  if (Array.isArray(lenke)) {
    return lenke.map(function (rad) {
      return Array.isArray(rad) ? rad.map(AIMPIRELENKE) : AIMPIRELENKE(rad);
    });
  }
  var t = String(lenke || '').trim();
  if (!t || !LENKE_OK.test(t)) return '';
  try {
    var k = kodeAv_(t);
    return k ? HUB + '?l=' + k : '';
  } catch (feil) { return ''; }
}

/** Skriver «Aimpire-lenke» som verdier for alle rader. Fjerner en gammel ARRAYFORMULA. */
function skrivLenker() {
  var ss = SpreadsheetApp.getActive();
  var ark = ss.getSheets().filter(erSoeknadsfane_)[0];
  if (!ark) throw new Error('Fant ikke søknadsfanen.');
  var siste = ark.getLastColumn();
  var tittel = ark.getRange(1, 1, 1, siste).getValues()[0];
  var cLenke = kolonne_(tittel, 'Partner link');
  var cA = tittel.indexOf(KOL_AIMPIRE) + 1;
  if (!cA) { cA = siste + 1; ark.getRange(1, cA).setValue(KOL_AIMPIRE); }
  var n = ark.getLastRow() - 1;
  if (n < 1) return;
  var lenker = ark.getRange(2, cLenke, n, 1).getValues();
  var ut = lenker.map(function (r) {
    var t = String(r[0] || '').trim();
    if (!LENKE_OK.test(t)) return [''];
    var k = kodeAv_(t);
    return [k ? HUB + '?l=' + k : ''];
  });
  ark.getRange(2, cA, n, 1).clearContent();   /* tar ogsaa bort en gammel formel i rad 2 */
  ark.getRange(2, cA, n, 1).setValues(ut);
}

/** Meny: send hver godkjent leder lenken sin (én gang per e-postadresse). */
function sendTilAlleGodkjente() {
  var ui = SpreadsheetApp.getUi();
  var ark = SpreadsheetApp.getActive().getSheets().filter(erSoeknadsfane_)[0];
  var siste = ark.getLastColumn();
  var tittel = ark.getRange(1, 1, 1, siste).getValues()[0];
  var cG = kolonne_(tittel, KOL_GODKJENT), cN = kolonne_(tittel, 'First name'), cL = kolonne_(tittel, 'Partner link'), cE = kolonne_(tittel, 'Email');
  var cS = tittel.indexOf('Language') + 1;
  var rader = ark.getRange(2, 1, Math.max(ark.getLastRow() - 1, 1), siste).getValues();
  var sett = {}, liste = [];
  rader.forEach(function (r) {
    var navn = String(r[cN - 1] || '').trim(), lenke = String(r[cL - 1] || '').trim(), epost = String(r[cE - 1] || '').trim().toLowerCase();
    if (r[cG - 1] !== true || !navnOk_(navn) || !LENKE_OK.test(lenke) || epost.indexOf('@') < 1) return;
    if (sett[epost]) return;   /* samme leder har soekt flere ganger: én e-post */
    sett[epost] = true;
    var sp = String(cS ? r[cS - 1] : 'en').trim().toLowerCase();
    liste.push({ navn: navn, epost: epost, lenke: HUB + '?l=' + kodeAv_(lenke), sp: TEKST_LAGRET[sp] ? sp : 'en' });
  });
  if (!liste.length) { ui.alert('Ingen godkjente ledere med gyldig lenke og e-post.'); return; }
  var svar = ui.alert('Send lenken til ' + liste.length + ' ledere?', liste.map(function (x) { return x.navn; }).join(', '), ui.ButtonSet.YES_NO);
  if (svar !== ui.Button.YES) return;
  var ok = 0, feil = [];
  liste.forEach(function (x) {
    if (MailApp.getRemainingDailyQuota() < 1) { feil.push(x.navn + ' (kvote)'); return; }
    var t = TEKST_LAGRET[x.sp];
    var tekst = t.body.replace('{NAME}', function () { return x.navn; }).replace('{LINK}', function () { return x.lenke; });
    try { MailApp.sendEmail(x.epost, t.subject, tekst, { name: 'Aimpire' }); ok++; }
    catch (e) { feil.push(x.navn + ' (' + String(e && e.message || e).slice(0, 40) + ')'); }
  });
  ui.alert('Sendt: ' + ok + (feil.length ? '\nIkke sendt: ' + feil.join(', ') : ''));
}
